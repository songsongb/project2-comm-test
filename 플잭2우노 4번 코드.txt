#include <Arduino.h>
#include <stdio.h>
#include <string.h>

// Arduino UNO R3. The relay's NO contact supplies the laser module.
constexpr uint8_t HUB_BIT_0 = 2;
constexpr uint8_t HUB_BIT_1 = 3;
constexpr uint8_t TEST_ACTIVE = 4;
constexpr uint8_t TEST_BUTTON = 6;
constexpr uint8_t PULSE_REFERENCE = 7;
constexpr uint8_t BUZZER = 8;
constexpr uint8_t RED = 9;
constexpr uint8_t GREEN = 10;
constexpr uint8_t BLUE = 11;
constexpr uint8_t RELAY = 12;

constexpr bool RELAY_ACTIVE_LOW = true;
constexpr bool BUZZER_ACTIVE_LOW = false;
constexpr bool RGB_COMMON_ANODE = false;
constexpr unsigned BUTTON_PULSE_MS = 150;
constexpr unsigned MIN_PULSE_MS = 50;
constexpr unsigned MAX_PULSE_MS = 400;
constexpr unsigned BETWEEN_PULSES_MS = 1050;
constexpr unsigned LASER_WARMUP_MS = 1000;
constexpr unsigned STABLE_INPUT_MS = 30;

enum TestPhase { IDLE, READY_DELAY, LASER_OFF, RECOVERY };
TestPhase phase = IDLE;
uint32_t phaseStarted;
uint32_t delayUntilNextPulse;
unsigned pulseWidth;
unsigned remaining;
unsigned pulseNumber;
bool cancelled;

uint8_t hubCandidate = 0;
uint8_t hubState = 0;
uint32_t hubCandidateSince;
bool buttonCandidate = false;
bool buttonPressed = false;
uint32_t buttonCandidateSince;
bool localDemo = false;
uint32_t demoStarted;

char command[40];
uint8_t commandLength = 0;
bool commandOverflow = false;

void setLaser(bool on) {
  digitalWrite(RELAY, on != RELAY_ACTIVE_LOW ? HIGH : LOW);
}

void setBuzzer(bool on) {
  digitalWrite(BUZZER, on != BUZZER_ACTIVE_LOW ? HIGH : LOW);
}

void setColor(uint8_t red, uint8_t green, uint8_t blue) {
  analogWrite(RED, RGB_COMMON_ANODE ? 255 - red : red);
  analogWrite(GREEN, RGB_COMMON_ANODE ? 255 - green : green);
  analogWrite(BLUE, RGB_COMMON_ANODE ? 255 - blue : blue);
}

void startTest(unsigned width, unsigned count, uint32_t now) {
  if (phase != IDLE || localDemo || hubState != 0) {
    Serial.println(F("BUSY_OR_HUB_NOT_NORMAL"));
    return;
  }
  pulseWidth = width;
  remaining = count;
  pulseNumber = 0;
  cancelled = false;
  phaseStarted = now;
  delayUntilNextPulse = LASER_WARMUP_MS;
  phase = READY_DELAY;
  digitalWrite(TEST_ACTIVE, HIGH);
  setLaser(true);
  Serial.println(F("TEST_STARTED"));
}

void finishTest() {
  phase = IDLE;
  digitalWrite(TEST_ACTIVE, LOW);
  Serial.println(cancelled ? F("TEST_ABORTED") : F("TEST_DONE"));
}

void stopTest(uint32_t now) {
  if (phase == IDLE) {
    return;
  }
  setLaser(true);
  digitalWrite(PULSE_REFERENCE, LOW);
  cancelled = true;
  remaining = 0;
  phaseStarted = now;
  delayUntilNextPulse = BETWEEN_PULSES_MS;
  phase = RECOVERY;
}

void serviceTest(uint32_t now) {
  if (phase != IDLE && !cancelled && hubState != 0) {
    stopTest(now);
  }
  switch (phase) {
    case IDLE:
      break;
    case READY_DELAY:
      if (now - phaseStarted >= delayUntilNextPulse) {
        ++pulseNumber;
        --remaining;
        phase = LASER_OFF;
        phaseStarted = now;
        digitalWrite(PULSE_REFERENCE, HIGH);
        setLaser(false);
      }
      break;
    case LASER_OFF:
      if (now - phaseStarted >= pulseWidth) {
        setLaser(true);
        digitalWrite(PULSE_REFERENCE, LOW);
        const uint32_t offMs = now - phaseStarted;
        phaseStarted = now;
        phase = RECOVERY;
        Serial.print(F("pulse,"));
        Serial.print(pulseNumber);
        Serial.print(',');
        Serial.println(offMs);
        // Vary spacing so short pulses do not repeatedly hit the same load phase.
        delayUntilNextPulse = BETWEEN_PULSES_MS + (pulseNumber * 73u) % 197u;
      }
      break;
    case RECOVERY:
      if (now - phaseStarted >= delayUntilNextPulse) {
        if (remaining) {
          phase = READY_DELAY;
          delayUntilNextPulse = 0;
        } else {
          finishTest();
        }
      }
      break;
  }
}

void serviceHub(uint32_t now) {
  const uint8_t raw = (digitalRead(HUB_BIT_0) == HIGH ? 1u : 0u)
                    | (digitalRead(HUB_BIT_1) == HIGH ? 2u : 0u);
  if (raw != hubCandidate) {
    hubCandidate = raw;
    hubCandidateSince = now;
  } else if (now - hubCandidateSince >= STABLE_INPUT_MS) {
    hubState = raw;
  }

  uint8_t shown = hubState;
  const uint32_t demoAge = now - demoStarted;
  if (localDemo && demoAge < 4000u && hubState == 0) {
    shown = demoAge / 1000u;
  } else {
    localDemo = false;
  }
  const bool blink = (now / 180u) % 2u == 0u;
  switch (shown) {
    case 0:
      setColor(0, 255, 0);
      setBuzzer(false);
      break;
    case 1:
      setColor(255, 160, 0);
      setBuzzer(false);
      break;
    case 2:
      setColor(blink ? 255 : 0, 0, 0);
      setBuzzer(blink);
      break;
    default:
      setColor(140, 0, 255);
      setBuzzer(false);
      break;
  }
}

void serviceButton(uint32_t now) {
  const bool raw = digitalRead(TEST_BUTTON) == LOW;
  if (raw != buttonCandidate) {
    buttonCandidate = raw;
    buttonCandidateSince = now;
  } else if (raw != buttonPressed && now - buttonCandidateSince >= STABLE_INPUT_MS) {
    buttonPressed = raw;
    if (buttonPressed) {
      if (phase == IDLE) {
        startTest(BUTTON_PULSE_MS, 1, now);
      } else {
        stopTest(now);
      }
    }
  }
}

void runCommand(uint32_t now) {
  int width, count;
  char extra;
  if (strcmp(command, "STOP") == 0) {
    stopTest(now);
  } else if (strcmp(command, "LASER ON") == 0 && phase == IDLE) {
    setLaser(true);
    Serial.println(F("LASER_ON"));
  } else if (strcmp(command, "LASER OFF") == 0 && phase == IDLE) {
    setLaser(false);
    Serial.println(F("LASER_OFF"));
  } else if (strcmp(command, "DEMO") == 0 && phase == IDLE && hubState == 0) {
    demoStarted = now;
    localDemo = true;
  } else if (sscanf(command, "TEST %d %d %c", &width, &count, &extra) == 2
             && width >= (int)MIN_PULSE_MS && width <= (int)MAX_PULSE_MS
             && count >= 1 && count <= 100) {
    startTest(width, count, now);
  } else {
    Serial.println(F("LASER ON | LASER OFF | DEMO | TEST 50..400 1..100 | STOP"));
  }
}

void serviceSerial(uint32_t now) {
  // Bound UART parsing per loop so it cannot starve the relay scheduler.
  for (uint8_t i = 0; i < 16 && Serial.available() > 0; ++i) {
    char ch = (char)Serial.read();
    if (ch == '\r' || ch == '\n') {
      if (commandOverflow) {
        Serial.println(F("COMMAND_TOO_LONG"));
      } else if (commandLength) {
        command[commandLength] = '\0';
        runCommand(now);
      }
      commandLength = 0;
      commandOverflow = false;
    } else if (commandLength < sizeof(command) - 1) {
      command[commandLength++] = ch;
    } else {
      commandOverflow = true;
    }
  }
}

void setup() {
  // Keep the relay inactive when changing the pin direction.
  setLaser(false);
  pinMode(RELAY, OUTPUT);
  setBuzzer(false);
  pinMode(BUZZER, OUTPUT);
  pinMode(RED, OUTPUT);
  pinMode(GREEN, OUTPUT);
  pinMode(BLUE, OUTPUT);
  pinMode(HUB_BIT_0, INPUT);
  pinMode(HUB_BIT_1, INPUT);
  pinMode(TEST_ACTIVE, OUTPUT);
  pinMode(PULSE_REFERENCE, OUTPUT);
  pinMode(TEST_BUTTON, INPUT_PULLUP);
  digitalWrite(TEST_ACTIVE, LOW);
  digitalWrite(PULSE_REFERENCE, LOW);
  setColor(0, 0, 0);
  Serial.begin(115200);
  setLaser(true);
  Serial.println(F("Laser Guard D ready. Enter HELP for commands."));
}

void loop() {
  const uint32_t now = millis();
  serviceHub(now);
  serviceTest(now);
  serviceButton(now);
  serviceSerial(now);
}
